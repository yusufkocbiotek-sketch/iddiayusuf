# Silinecek başlangıçlar
SILINECEK = ("oyuncu", "100.00", "takım")


def key_donustur(oranlar):
    """Tüm key'leri dönüştür + istenmeyenleri sil"""
    yeni = {}
    silinen = 0
    donusturulen = 0

    for key, value inoranlar.items():
        k = key.strip().lower()

        # İstenmeyenleri sil
        if k.startswith(SILINECEK):
            silinen += 1
            continue

        # Key dönüştür
        yeni_key = KEY_MAP.get(key, key)
        if yeni_key != key:
            donusturulen += 1

        yeni[yeni_key] = value

    return yeni, silinen, donusturulen


def sirala_indexle(matches):
    """Tarihe göre sırala ve index yaz"""
    sirali = sorted(matches, key=lambda x: (x.get("tarih", ""), x.get("saat", "00:00")))
    for i, mac in enumerate(sirali, 1):
        mac["index"] = i
    return sirali
