Evet, kesinlikle yapabiliriz! Bu aslında profesyonel bahis analiz sitelerinin yaptığı **"Value Bet / Değer Bahsi"** mantığıdır. 

### Nasıl Çalışır?

Her maçın oranlarından **zımni olasılık** (implied probability) hesaplanır:

```
Zımni Olasılık = 1 / Oran

Örnek:
  MS 1 (Galatasaray kazanır): 1.50 → 1/1.50 = %66.7
  KG Var (Karşılıklı gol):    1.80 → 1/1.80 = %55.6
  
  2'li Kombine Olasılık: %66.7 × %55.6 = %37.1  (Bu %75'in ALTINDA, seçilmez)
  
  Başka örnek:
  MS 1: 1.25 → %80
  KG Yok: 1.40 → %71.4
  
  2'li Kombine: %80 × %71.4 = %57.1  (Bu da %75'in altında)
```

**Aslında %75 üstü 2'li kombinasyon bulmak çok zor.** Bunun yerine daha gerçekçi bir sistem kurabiliriz:

| Strateji | Mantık | Eşik |
|---|---|---|
| **Tekli Güçlü** | Tek bir oranın olasılığı %75+ | MS 1 @ 1.33 altı |
| **2'li Kombine** | İki oranın çarpımı > %50 | Esnek eşik |
| **Skor Tahmini** | MS + KG + Alt/Üst birlikte | Ağırlıklı puan |

### Sistem Nasıl Kurulur?

**1. Python Analiz Scripti (`oran_analiz.py`):** `mac.json`'daki her maçın oranlarını analiz eder ve maçlara `"analiz"` etiketi ekler.

**2. Frontend (Siten):** `mac.json`'daki `"analiz"` etiketine göre maçları renklendirir (kırmızı, yeşil, sarı).

Şimdi ikisini de yazıyorum. Önce Python analiz scripti:

### `oran_analiz.py` (Yeni Dosya Oluştur)

```python
import json
import os
import datetime

MAC_JSON = "public/data/mac.json"

# ============================
# EŞİKLER (İstersen değiştir)
# ============================
ESIK_TEKLI_GUC = 0.75       # Tekli oranın zımni olasılığı %75+
ESIK_2LI_KOMBIN = 0.50      # 2'li kombine olasılık %50+
ESIK_3LU_KOMBIN = 0.40      # 3'lü kombine olasılık %40+
SKORLAMA_ESIK = 3           # Toplam puan 3+ ise "Güçlü Maç"

# ============================
# ORAN ANALİZ FONKSİYONLARI
# ============================
def zimni_olasilik(oran):
    """Orandan zımni olasılık hesapla (0-1 arası)"""
    try:
        if oran and float(oran) > 1.0:
            return 1 / float(oran)
    except:
        pass
    return 0

def mac_analiz_et(mac):
    """Bir maçın tüm oranlarını analiz et ve puanla"""
    oranlar = mac.get("oranlar", {})
    if not oranlar:
        return None
    
    analiz = {
        "guclu_favori": False,
        "kg_var_yuksek": False,
        "ust_25_yuksek": False,
        "toplam_puan": 0,
        "oneri": "",
        "renk": "normal",
        "kombinasyonlar": []
    }
    
    puan = 0
    
    # ---- 1. TEKLI GÜÇLÜ FAVORİ ----
    ms1 = zimni_olasilik(oranlar.get("Maç Sonucu_1"))
    ms0 = zimni_olasilik(oranlar.get("Maç Sonucu_0"))
    ms2 = zimni_olasilik(oranlar.get("Maç Sonucu_2"))
    
    en_guclu = max(ms1, ms0, ms2)
    
    if en_guclu >= ESIK_TEKLI_GUC:
        analiz["guclu_favori"] = True
        puan += 2
        if en_guclu >= 0.85:
            puan += 1  # Süper favori bonusu
    
    # ---- 2. KARŞILIKLI GOL ----
    kg_var = zimni_olasilik(oranlar.get("Karşılıklı Gol_Var"))
    kg_yok = zimni_olasilik(oranlar.get("Karşılıklı Gol_Yok"))
    
    if kg_var >= 0.65:
        analiz["kg_var_yuksek"] = True
        puan += 1
    
    if kg_yok >= 0.65:
        puan += 1
    
    # ---- 3. ALT/ÜST ----
    ust_25 = zimni_olasilik(oranlar.get("Alt/Üst 2.5_Üst"))
    alt_25 = zimni_olasilik(oranlar.get("Alt/Üst 2.5_Alt"))
    
    if ust_25 >= 0.60:
        analiz["ust_25_yuksek"] = True
        puan += 1
    
    if alt_25 >= 0.60:
        puan += 1
    
    # ---- 4. İY SONUÇ ----
    iy1 = zimni_olasilik(oranlar.get("İY Sonuç_1"))
    iy0 = zimni_olasilik(oranlar.get("İY Sonuç_0"))
    iy2 = zimni_olasilik(oranlar.get("İY Sonuç_2"))
    
    en_guclu_iy = max(iy1, iy0, iy2)
    if en_guclu_iy >= 0.55:
        puan += 1
    
    # ---- 5. 2'Lİ KOMBİNASYONLAR ----
    kombinasyonlar = []
    
    # MS + KG kombinasyonları
    for ms_key, ms_prob in [("MS 1", ms1), ("MS 0", ms0), ("MS 2", ms2)]:
        for kg_key, kg_prob in [("KG Var", kg_var), ("KG Yok", kg_yok)]:
            if ms_prob > 0.3 and kg_prob > 0.3:
                kombin_prob = ms_prob * kg_prob
                kombinasyonlar.append({
                    "bahis": f"{ms_key} + {kg_key}",
                    "olasilik": round(kombin_prob * 100, 1),
                    "guclu": kombin_prob >= ESIK_2LI_KOMBIN
                })
                if kombin_prob >= ESIK_2LI_KOMBIN:
                    puan += 1
    
    # MS + Alt/Üst kombinasyonları
    for ms_key, ms_prob in [("MS 1", ms1), ("MS 0", ms0), ("MS 2", ms2)]:
        for au_key, au_prob in [("Üst 2.5", ust_25), ("Alt 2.5", alt_25)]:
            if ms_prob > 0.3 and au_prob > 0.3:
                kombin_prob = ms_prob * au_prob
                kombinasyonlar.append({
                    "bahis": f"{ms_key} + {au_key}",
                    "olasilik": round(kombin_prob * 100, 1),
                    "guclu": kombin_prob >= ESIK_2LI_KOMBIN
                })
                if kombin_prob >= ESIK_2LI_KOMBIN:
                    puan += 1
    
    # En iyi kombinasyonları sırala
    kombinasyonlar.sort(key=lambda x: x["olasilik"], reverse=True)
    analiz["kombinasyonlar"] = kombinasyonlar[:5]  # En iyi 5 tanesini sakla
    
    # ---- 6. SONUÇ VE RENK ----
    analiz["toplam_puan"] = puan
    
    if puan >= 5:
        analiz["renk"] = "kirmizi"       # 🔴 Çok güçlü maç
        analiz["oneri"] = "🔥 SÜPER FIRSAT"
    elif puan >= 3:
        analiz["renk"] = "turuncu"       # 🟠 Güçlü maç
        analiz["oneri"] = "⚡ GÜÇLÜ MAÇ"
    elif puan >= 2:
        analiz["renk"] = "sari"          # 🟡 Dikkat çekici
        analiz["oneri"] = "📌 DİKKAT"
    else:
        analiz["renk"] = "normal"        # ⚪ Sıradan
        analiz["oneri"] = ""
    
    return analiz

def main():
    print("=" * 60)
    print("📊 ORAN ANALİZ SİSTEMİ")
    print("=" * 60)
    
    if not os.path.exists(MAC_JSON):
        print("❌ mac.json dosyası bulunamadı!")
        return
    
    with open(MAC_JSON, "r", encoding="utf-8") as f:
        data = json.load(f)
    
    maclar = data.get("matches", [])
    if not maclar:
        print("❌ Maç bulunamadı!")
        return
    
    kirmizi = 0
    turuncu = 0
    sari = 0
    normal = 0
    
    for mac in maclar:
        analiz = mac_analiz_et(mac)
        if analiz:
            mac["analiz"] = analiz
            
            if analiz["renk"] == "kirmizi":
                kirmizi += 1
                print(f"🔴 [{mac.get('saat','')}] {mac.get('ev_sahibi','')} - {mac.get('deplasman','')} "
                      f"| Puan: {analiz['toplam_puan']} | {analiz['oneri']}")
                if analiz["kombinasyonlar"]:
                    en_iyi = analiz["kombinasyonlar"][0]
                    print(f"   💰 En iyi kombin: {en_iyi['bahis']} = %{en_iyi['olasilik']}")
            
            elif analiz["renk"] == "turuncu":
                turuncu += 1
                print(f"🟠 [{mac.get('saat','')}] {mac.get('ev_sahibi','')} - {mac.get('deplasman','')} "
                      f"| Puan: {analiz['toplam_puan']} | {analiz['oneri']}")
            
            elif analiz["renk"] == "sari":
                sari += 1
            else:
                normal += 1
    
    # Dosyaya kaydet
    data["updated"] = datetime.datetime.now().isoformat()
    with open(MAC_JSON, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
    
    print(f"\n{'='*60}")
    print(f"📊 ANALİZ SONUÇLARI")
    print(f"   🔴 Süper Fırsat (Kırmızı): {kirmizi}")
    print(f"   🟠 Güçlü Maç (Turuncu):    {turuncu}")
    print(f"   🟡 Dikkat (Sarı):           {sari}")
    print(f"   ⚪ Normal:                   {normal}")
    print(f"   📊 Toplam:                   {len(maclar)}")
    print(f"{'='*60}")
    print(f"\n💾 mac.json dosyasına 'analiz' etiketleri eklendi!")
    print(f"   Artık sitende maçları renklendirebilirsin!")

if __name__ == "__main__":
    main()
```
