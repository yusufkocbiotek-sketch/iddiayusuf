import json
import pandas as pd
from pandas import json_normalize

# JSON dosyasını oku
with open('public/data/mac.json', 'r', encoding='utf-8') as f:
    data = json.load(f)

# Maç listesini al
matches = data.get('matches', [])

# 🔑 EN ÖNEMLİ KISIM: İç içe geçmiş "oranlar" verisini düzleştirip sütunlara çevir
df = json_normalize(matches)

# Gerekli temel sütunları belirle (Varsa gelecek, yoksa hata vermeyecek)
temel_sutunlar = [
    'tarih', 'saat', 'lig', 'ev_sahibi', 'deplasman', 
    'durum', 'skor_ev', 'skor_dep', 'skor_1y_ev', 'skor_1y_dep'
]

# Oran sütunları (JSON'daki isimleri neyse onlar yazıldı, senin kullandığın harflerle eşleşir)
oran_sutunlari = [
    'oranlar.ms1',    # J sütunu
    'oranlar.ms0',    # K sütunu
    'oranlar.ms2',    # M sütunu
    'oranlar.kg_var', # O sütunu
    'oranlar.kg_yok', # P sütunu
    'oranlar.alt',    # H sütunu
    'oranlar.ust'     # I sütunu
]

# Mevcut olan tüm sütunları birleştir, olmayanları otomatik atla
secilecek_sutunlar = []
for sutun in temel_sutunlar + oran_sutunlari:
    if sutun in df.columns:
        secilecek_sutunlar.append(sutun)

# Sadece seçilen sütunları tut
df = df[secilecek_sutunlar]

# Sütun isimlerini Excel'de senin kullandığın kısa isimlere çevir (İsteğe bağlı, daha rahat çalışmak için)
isim_degistir = {
    'oranlar.ms1': 'J',
    'oranlar.ms0': 'K',
    'oranlar.ms2': 'M',
    'oranlar.kg_var': 'O',
    'oranlar.kg_yok': 'P',
    'oranlar.alt': 'H',
    'oranlar.ust': 'I',
    'skor_ev': 'G_EV',
    'skor_dep': 'G_DEP',
    'skor_1y_ev': 'F_EV',
    'skor_1y_dep': 'F_DEP'
}
df = df.rename(columns=isim_degistir)

# Skoru tek sütunda birleştir (Senin G sütunun gibi "2-1" formatında)
if 'G_EV' in df.columns and 'G_DEP' in df.columns:
    df['G'] = df['G_EV'].astype(str) + "-" + df['G_DEP'].astype(str)
if 'F_EV' in df.columns and 'F_DEP' in df.columns:
    df['F'] = df['F_EV'].astype(str) + "-" + df['F_DEP'].astype(str)

# Excel olarak kaydet
output_file = 'mac_raporu.xlsx'
df.to_excel(output_file, index=False)

print(f"✅ Başarılı! {len(df)} maç '{output_file}' dosyasına kaydedildi.")
print("📌 Tüm oranlar ve skorlar ayrı sütunlara yazıldı.")
