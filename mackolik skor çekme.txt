def tarihleri_olustur(baslangic_str, bitis_str):
    """Farklı tarih formatlarını algılar ve uygun hale getirir"""
    def str_to_tarih(t):
        t = str(t).strip()
        if "-" in t: # 2026-05-20 formatı
            y, a, g = map(int, t.split('-'))
            return datetime.date(y, a, g)
        elif "/" in t:
            parcalar = list(map(int, t.split('/')))
            if len(parcalar) == 2: # 17/05 formatı
                g, a = parcalar
                return datetime.date(2026, a, g)
            elif len(parcalar) == 3: # 20/05/2026 formatı
                g, a, y = parcalar
                return datetime.date(y, a, g)
        return None

    baslangic = str_to_tarih(baslangic_str)
    bitis = str_to_tarih(bitis_str)
    
    if not baslangic or not bitis:
        print("❌ Tarih formatı hatalı! Örnek: 17/05 veya 20/05/2026 veya 2026-05-20")
        return []

    tarihler = []
    while baslangic <= bitis:
        tarihler.append(f"{baslangic.day:02d}/{baslangic.month:02d}") # Mackolik formatı: GÜN/AY
        baslangic += datetime.timedelta(days=1)
    return tarihler

def tarihi_esit_kabul_et(t1, t2):
    if not t1 or not t2: return False
    def duzelt(t):
        t = str(t).strip()
        t = t.replace(":", "").replace("-", "").replace("/", "").replace(" ", "")
        if len(t) >= 8: t = t[:8]
        return t
    return duzelt(t1) == duzelt(t2)

# 🔐 İSİM TEMİZLEME - ✅ ÇOK ESNEK HALE GELDİ, HİÇBİR İSMİ ELEMİYOR

def akilli_isim_temizle(isim):
    if not isim: return ""
    isim = isim.lower().strip()
    tr_map = str.maketrans("çğıöşüâêîôûáéíóúñðßçşğ", "cgiosuaeiouaeiounbscsg")
    isim = isim.translate(tr_map)
    # Sadece gereksiz karakterleri sil, isimlerin kendisine DOKUNMA
    isim = re.sub(r'[.\-_,:0-9]', '', isim)
    # Sadece çok yaygın ekleri çıkar, isim kısaltmalarını bırak
    gereksiz = ['spor', 'kulubu', 'kulübü', 'team']
    for ek in gereksiz:
        isim = isim.replace(ek, '')
    isim = re.sub(r'\s+', ' ', isim).strip()
    
    # ❌ ARTIK "GEÇERSİZ" YOK, EN AZ 2 HARF OLAN HERKESİ KABUL ET
    if len(isim) < 2: 
        return ""
    return isim

def benzerlik_orani(a, b):
    if not a or not b: return 0.0
    a_temiz = akilli_isim_temizle(a)
    b_temiz = akilli_isim_temizle(b)
    if a_temiz == b_temiz: return 1.0
    if a_temiz in b_temiz or b_temiz in a_temiz: return 0.8
    return round(SequenceMatcher(None, a_temiz, b_temiz).ratio(), 2)

# 🚨 SKOR MANTIK KONTROLÜ - ✅ UYARILARI KALDIRDIK
# =============================================================================
def skor_mantikli_mi(ev, dep, iy_ev, iy_dep, durum, ev_isim="", dep_isim=""):
    if not KURAL_SKOR_KONTROL:
        return True, "Kontrol Kapalı", True

    # 1. İLK YARI > MAÇ SONU DURUMU: Artık uyar, ama KAYDET
    if iy_ev > ev or iy_dep > dep:
        if MANTIK_HATASI_DUZELT:
            return True, f"ℹ️ Bilgi | {ev_isim}-{dep_isim} | Veri geç gelmiş, kısmi kaydedildi", False
        else:
            return False, f"❌ HATA | {ev_isim}-{dep_isim} | İmkansız Skor", False

    # 2. Tümü 0-0 durumu: KAYDET
    if KURAL_SIFIR_KONTROL and durum == "bitti":
        if ev == 0 and dep == 0 and iy_ev == 0 and iy_dep == 0:
            return True, f"ℹ️ Bilgi | {ev_isim}-{dep_isim} | 0-0 Beraberlik", True

    return True, "✅ Uygun Veri", True

def load_mac_json():
    try:
        if not MAC_JSON_PATH.exists():
            print(f"❌ HATA: {MAC_JSON_PATH} bulunamadı!")
            return None
        with open(MAC_JSON_PATH, 'r', encoding='utf-8') as f:
            veri = json.load(f)
        if isinstance(veri, dict) and "matches" in veri:
            print(f"📖 Dosya Okundu | Toplam: {len(veri['matches'])} maç.")
            return veri 
        else:
            print("❌ HATA: Dosya formatı uyumsuz.")
            return None
    except Exception as e:
        print(f"❌ OKUMA HATASI: {e}")
        return None

def save_mac_json(veri):
    try:
        yedek = MAC_JSON_PATH.with_name("mac_json_yedek_guvenli.json")
        with open(yedek, 'w', encoding='utf-8') as f:
            json.dump(veri, f, ensure_ascii=False, indent=2)
        with open(MAC_JSON_PATH, 'w', encoding='utf-8') as f:
            json.dump(veri, f, ensure_ascii=False, indent=2, default=str)
        print(f"💾 Kayıt Başarılı | Yedek: {yedek.name}")
        print("🔒 Oran, Lig, Saat, Kodlar KORUNDU.")
    except Exception as e:
        print(f"❌ KAYDETME HATASI: {e}")

def git_islemlerini_yap():
    print("\n" + "="*70)
    print("🚀 GİT İŞLEMLERİ BAŞLATILDI...")
    print("="*70)
    try:
        os.chdir(BASE_DIR)
        durum_cikisi = subprocess.run(["git","status"], capture_output=True, text=True).stdout
        if "nothing to commit" in durum_cikisi:
            print("ℹ️ Değişiklik yok.")
            return False
        subprocess.run(["git","add","."], check=True)
        subprocess.run(["git","commit","-m",f"[OTOMATİK] {BASLANGIC_TARIHI}-{BITIS_TARIHI} | Düzeltme Yapıldı"], check=True)
        subprocess.run(["git","push","origin",GIT_BRANCH_NAME], check=True)
        print("✅ GİT BAŞARILI!")
        return True
    except Exception as e:
        print(f"❌ GİT HATASI: {e}")
        return False

def get_skorlar_tek_gun(hedef_tarih):
    print(f"\n📅 İŞLENİYOR: {hedef_tarih} | 🛡️ SAĞLAM MOD AKTİF")
    skor_listesi = []
    gorulen = set()
    atlanan_sayisi = 0
    kismi_veri_sayisi = 0

    chrome_options = Options()
    chrome_options.add_argument("--start-maximized")
    chrome_options.add_argument("--no-sandbox")
    chrome_options.add_argument("--disable-dev-shm-usage")
    chrome_options.add_argument("--log-level=3")
    chrome_options.add_argument("user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36")
    chrome_options.add_argument("--disable-blink-features=AutomationControlled")
    chrome_options.add_experimental_option("excludeSwitches", ["enable-automation"])
    chrome_options.add_experimental_option("useAutomationExtension", False)

    driver = webdriver.Chrome(options=chrome_options)
    driver.execute_script("Object.defineProperty(navigator, 'webdriver', {get: () => undefined})")

    try:
        print(f"🌐 Site: {BASE_LINK}")
        driver.get(BASE_LINK)
        print(f"⌛ Sayfa Yüklenmesi Bekleniyor... ({SAYFA_YUKLEME_BEKLEME} sn)")
        time.sleep(SAYFA_YUKLEME_BEKLEME // 2)

        # Tarih Seçimi - ✅ Geliştirilmiş seçici
        try:
            tarihler = driver.find_elements(By.CSS_SELECTOR, "span.widget-dateslider__day-date, div.date-item")
            bulundu = False
            for el in tarihler:
                metin = el.text.strip()
                if hedef_tarih in metin:
                    driver.execute_script("arguments[0].scrollIntoView({block:'center'});", el)
                    time.sleep(2)
                    driver.execute_script("arguments[0].click();", el)
                    print(f"✅ Tarih Seçildi: {hedef_tarih}")
                    bulundu = True
                    time.sleep(SAYFA_YUKLEME_BEKLEME // 2)
                    break
            if not bulundu:
                print(f"⚠️ {hedef_tarih} sayfada bulunamadı, atlanıyor.")
                return []
        except Exception as e:
            print(f"⚠️ Tarih seçim hatası: {e}")
            return []

        # 📜 SAYFAYI YAVAŞÇA İNDİR
        print("📜 Sayfa yavaşça kaydırılıyor, tüm maçlar yüklensin...")
        for _ in range(120): 
            driver.execute_script("window.scrollBy(0, 600);")
            time.sleep(0.4)
        time.sleep(5)

        mac_satirlari = driver.find_elements(By.CSS_SELECTOR, "div.match-row, div.row-table__row, tr.match, li.match-item")
        print(f"🔍 Toplam Satır: {len(mac_satirlari)} | 🔁 {TEKRAR_OKUMA_SAYISI} Kere Kontrol Edilecek")

        gun, ay = hedef_tarih.split('/')
        hedef_tarih_iso = f"2026-{ay}-{gun}"
        gecerli_sayisi = 0

        for satir_index, satir in enumerate(mac_satirlari):
            try:
                # Takım İsimleri - ✅ Tüm olası seçiciler
                isimler = satir.find_elements(By.CSS_SELECTOR, 
                    "span.match-row__team-name-text, span.team-name, div.match-row__team-name, span.name, td.team, span.participant-name")
                if len(isimler) < 2: continue

                ev_isim = isimler[0].text.strip()
                dep_isim = isimler[1].text.strip()

                # Tekrar edenleri engelle
                kimlik = f"{akilli_isim_temizle(ev_isim)}-{akilli_isim_temizle(dep_isim)}"
                if kimlik in gorulen and kimlik != "-": 
                    continue
                gorulen.add(kimlik)

                if len(ev_isim) < 2 or len(dep_isim) < 2:
                    atlanan_sayisi +=1
                    continue

                # 📊 VERİ OKUMA - Çoklu Kontrol
                en_iyi_sonuc = {"s_ev":0, "s_dep":0, "iy_ev":0, "iy_dep":0, "durum":"baslamadi"}

                for tur in range(TEKRAR_OKUMA_SAYISI):
                    # Skorlar
                    try:
                        skor_el = satir.find_elements(By.CSS_SELECTOR, 
                            "span.match-row__score-text, span.score, div.score, td.score, span.result")
                        if len(skor_el)>=2:
                            s1 = skor_el[0].text.strip()
                            s2 = skor_el[1].text.strip()
                            if s1.isdigit(): en_iyi_sonuc["s_ev"] = int(s1)
                            if s2.isdigit(): en_iyi_sonuc["s_dep"] = int(s2)
                    except: pass

                    # İlk Yarı
                    try:
                        iy_el = satir.find_element(By.CSS_SELECTOR, 
                            "div.match-row__half-time-score, div.half-time, span.ht, td.ht, span.halftime")
                        rakamlar = re.findall(r'\d+', iy_el.text)
                        if len(rakamlar) == 2:
                            en_iyi_sonuc["iy_ev"] = int(rakamlar[0])
                            en_iyi_sonuc["iy_dep"] = int(rakamlar[1])
                    except: pass

                    try:
                        st_el = satir.find_element(By.CSS_SELECTOR, 
                            "a.match-row__status, div.match-row__status, span.status, td.status, span.match-status")
                        st_yazi = st_el.text.strip().upper()
                        if any(k in st_yazi for k in ["MS", "FİNAL", "BİTTİ", "SONUÇ"]):
                            en_iyi_sonuc["durum"] = "bitti"
                        elif any(k in st_yazi for k in ["CANLI", "DEVAM", "'", "DK"]):
                            en_iyi_sonuc["durum"] = "devam ediyor"
                    except:
                        if en_iyi_sonuc["s_ev"] > 0 or en_iyi_sonuc["s_dep"] > 0:
                            en_iyi_sonuc["durum"] = "devam ediyor"

                    time.sleep(TEKRARLAR_ARASI_BEKLEME)

                # Sonuçları al
                s_ev = en_iyi_sonuc["s_ev"]
                s_dep = en_iyi_sonuc["s_dep"]
                iy_ev = en_iyi_sonuc["iy_ev"]
                iy_dep = en_iyi_sonuc["iy_dep"]
                durum = en_iyi_sonuc["durum"]

                # ✅ KONTROL: Artık çok az atlama var
                mantikli, mesaj, _ = skor_mantikli_mi(s_ev, s_dep, iy_ev, iy_dep, durum, ev_isim, dep_isim)
                if not mantikli:
                    atlanan_sayisi += 1
                    continue

                # ✅ KAYDET
                skor_listesi.append({
                    "tarih": hedef_tarih_iso,
                    "ev_sahibi": ev_isim,
                    "deplasman": dep_isim,
                    "skor_ev": s_ev,
                    "skor_dep": s_dep,
                    "skor_1y_ev": iy_ev,
                    "skor_1y_dep": iy_dep,
                    "durum": durum
                })
                gecerli_sayisi += 1
                print(f"✅ KAYIT | {ev_isim} - {dep_isim} | MS:{s_ev}-{s_dep} | İY:{iy_ev}-{iy_dep} | {durum}")

            except Exception as satir_hata:
                atlanan_sayisi += 1
                continue

    except Exception as ana_hata:
        print(f"❌ Çekim Ana Hatası: {ana_hata}")
    finally:
        if driver:
            driver.quit()

    print(f"📅 {hedef_tarih} Özeti: Toplam Geçerli: {gecerli_sayisi} | Kısmi Veri: {kismi_veri_sayisi} | Tamamen Atlanan: {atlanan_sayisi}")
    return skor_listesi

# 🧠 GÜNCELLEME MOTORU - ✅ VERİ KORUMA KURALI %100 AKTİF
# =============================================================================
def skorlari_eslestir_ve_guncelle(mevcut_yapi, tum_veriler):
    mac_listesi = mevcut_yapi.get("matches", [])
    if not mac_listesi or not tum_veriler:
        return 0

    guncelleme_sayisi = 0
    eslesen_indexler = set()

    for y_veri in tum_veriler:
        y_tarih = y_veri["tarih"]
        y_ev = y_veri["ev_sahibi"]
        y_dep = y_veri["deplasman"]

        en_uygun = []

        for i, mac in enumerate(mac_listesi):
            if i in eslesen_indexler: continue

            m_tarih = mac.get("tarih", "")
            m_ev = mac.get("ev_sahibi", "")
            m_dep = mac.get("deplasman", "")

            if not tarihi_esit_kabul_et(m_tarih, y_tarih):
                continue

            o1 = benzerlik_orani(m_ev, y_ev) + benzerlik_orani(m_dep, y_dep)
            o2 = benzerlik_orani(m_ev, y_dep) + benzerlik_orani(m_dep, y_ev)

            if o1 >= ESLESME_SEVIYESI or o2 >= ESLESME_SEVIYESI:
                en_uygun.append( (-max(o1,o2), i, (o2>o1)) )

        if en_uygun:
            en_uygun.sort()
            _, index, ters_mi = en_uygun[0]
            eslesen_indexler.add(index)

            mac = mac_listesi[index]

            s_ev, s_dep = (y_veri["skor_ev"], y_veri["skor_dep"]) if not ters_mi else (y_veri["skor_dep"], y_veri["skor_ev"])
            iy_ev, iy_dep = (y_veri["skor_1y_ev"], y_veri["skor_1y_dep"]) if not ters_mi else (y_veri["skor_1y_dep"], y_veri["skor_1y_ev"])

            degisiklik_var = False

            # 🛡️ KORUMA: Eski veri DOLUYSA ve yeni veri 0 İSE DEĞİŞTİRME!
            if mac.get("skor_ev", 0) == 0 and s_ev > 0:
                mac["skor_ev"] = s_ev
                degisiklik_var = True
            if mac.get("skor_dep", 0) == 0 and s_dep > 0:
                mac["skor_dep"] = s_dep
                degisiklik_var = True

            # İlk yarı ve durum her zaman güncellenir
            if mac.get("skor_1y_ev") != iy_ev:
                mac["skor_1y_ev"] = iy_ev
                degisiklik_var = True
            if mac.get("skor_1y_dep") != iy_dep:
                mac["skor_1y_dep"] = iy_dep
                degisiklik_var = True
            if mac.get("durum") != y_veri["durum"]:
                mac["durum"] = y_veri["durum"]
                degisiklik_var = True

            if degisiklik_var:
                guncelleme_sayisi += 1
                print(f"✅ GÜNCELLEME | {mac['ev_sahibi']} - {mac['deplasman']} | MS:{mac['skor_ev']}-{mac['skor_dep']} | İY:{mac['skor_1y_ev']}-{mac['skor_1y_dep']} | {mac['durum']}")

    return guncelleme_sayisi
