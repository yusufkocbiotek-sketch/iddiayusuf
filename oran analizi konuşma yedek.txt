### Şimdi JavaScript Kısmı Gerekli

HTML tamam, ama tabloyu dolduracak JavaScript kodunu da eklememiz lazım. 

HTML dosyasının en altındaki `<script>` etiketinin **içindeki tüm kodları** kopyala ve buraya yapıştır. *(Çok uzunsa, sadece `<script>` ile `</script>` arasındaki kodu gönder).*

JS kodunu gördüğüm an, `mainTab(2)` fonksiyonunu ve `akilliKombinasyonFiltrele()` fonksiyonunu yazıp sana vereceğim! 🚀

Mükemmel! HTML kısmı **%100 doğru** olmuş. `footer` ve `<script>` satırları tam yerinde. 👏

Şimdi **JavaScript kısmını** yapalım:

### JS Değişikliği:

`<script>` etiketinin içinde, en alt kısımlarda şu **eski** fonksiyonları bul:

```javascript
function akilliKombinasyonFiltrele() { ... }
function akilliTabloRender(maclar) { ... }
const orjMainTab = mainTab;
mainTab = function(i) { ... }
```

*(Bu fonksiyonlar en altta yer alıyor. İçlerinde `akilliArama`, `akilliMinYuzde`, `akilliTabloBody` gibi kelimeler geçen blokları bul.)*

Bu **eski blokları tamamen sil** ve yerine benim verdiğim **yeni** `kombineKesifYap()`, `kombineKesifTemizle()` ve `mainTab` fonksiyonlarını yapıştır.

**Dikkat:** En alttaki şu iki satıra **dokunma**, onlar aynen kalsın:

```javascript
yukle();
setTimeout(kombine2InitSecenekler,1000);
```

Yeni JS kodlarını bu iki satırın **hemen üstüne** ekle.

### Değişiklik 2: JavaScript

`</script>` satırının hemen üstündeki **eski** `kombineKesif` veya `akilliKombinasyon` fonksiyonlarını **tamamen sil** (varsa). Sonra `yukle();` satırının **hemen üstüne** şunu ekle:

```javascript
/* ============================================ */
/* 3. SEKME: KOMBİNE KEŞİF                      */
/* ============================================ */

function kesifDogruMu(m, combo) {
  if (!m || m.durum !== "bitti") return false;
  const se = Number(m.skor_ev || 0), sd = Number(m.skor_dep || 0);
  const tg = se + sd;
  const kg = se > 0 && sd > 0;
  const ms = se > sd ? "1" : (se === sd ? "0" : "2");

  if (combo === "MS1") return ms === "1";
  if (combo === "MS0") return ms === "0";
  if (combo === "MS2") return ms === "2";
  if (combo === "ALT25") return tg < 2.5;
  if (combo === "UST25") return tg > 2.5;
  if (combo === "KGV") return kg;
  if (combo === "KGY") return !kg;
  if (combo === "MS1_KGV") return ms === "1" && kg;
  if (combo === "MS1_KGY") return ms === "1" && !kg;
  if (combo === "MS1_UST25") return ms === "1" && tg > 2.5;
  if (combo === "MS1_ALT25") return ms === "1" && tg < 2.5;
  if (combo === "MS2_KGV") return ms === "2" && kg;
  if (combo === "MS2_KGY") return ms === "2" && !kg;
  if (combo === "MS2_UST25") return ms === "2" && tg > 2.5;
  if (combo === "MS2_ALT25") return ms === "2" && tg < 2.5;
  if (combo === "MS0_KGV") return ms === "0" && kg;
  if (combo === "MS0_UST25") return ms === "0" && tg > 2.5;
  if (combo === "MS0_ALT25") return ms === "0" && tg < 2.5;
  return false;
}

function kesifAnaMarket(combo) {
  if (combo.startsWith("MS1")) return "Maç Sonucu_1";
  if (combo.startsWith("MS2")) return "Maç Sonucu_2";
  if (combo.startsWith("MS0")) return "Maç Sonucu_0";
  if (combo.includes("ALT25")) return "Alt/Üst 2.5_Alt";
  if (combo.includes("UST25")) return "Alt/Üst 2.5_Üst";
  if (combo.includes("KGV")) return "Karşılıklı Gol_Var";
  if (combo.includes("KGY")) return "Karşılıklı Gol_Yok";
  return null;
}

function kesifYuzdeHesapla(m, combo) {
  const anaMarket = kesifAnaMarket(combo);
  if (!anaMarket || !m.oranlar || !m.oranlar[anaMarket]) return null;

  const hedefOran = parseFloat(m.oranlar[anaMarket]);
  if (isNaN(hedefOran)) return null;

  const tolerans = 0.15;
  const altSinir = hedefOran - tolerans;
  const ustSinir = hedefOran + tolerans;

  const hepsi = tumMaclar.concat(gecmisMaclar);
  let benzerMaclar = hepsi.filter(bm => {
    if (bm.durum !== "bitti") return false;
    if (!bm.oranlar || !bm.oranlar[anaMarket]) return false;
    const bmOran = parseFloat(bm.oranlar[anaMarket]);
    return !isNaN(bmOran) && bmOran >= altSinir && bmOran <= ustSinir;
  });

  if (benzerMaclar.length < 3) return null;

  let tutan = 0;
  benzerMaclar.forEach(bm => {
    if (kesifDogruMu(bm, combo)) tutan++;
  });

  return {
    yuzde: Math.round((tutan / benzerMaclar.length) * 100),
    tutan: tutan,
    toplam: benzerMaclar.length
  };
}

function kombineKesifFiltrele() {
  const arama = (document.getElementById("kesifArama")?.value || "").toLowerCase();
  const tarih = document.getElementById("kesifTarih")?.value || "";
  const lig = document.getElementById("kesifLig")?.value || "";
  const durum = document.getElementById("kesifDurum")?.value || "";
  const combo = document.getElementById("kesifKombinasyon")?.value || "";
  const minYuzde = parseInt(document.getElementById("kesifMinYuzde")?.value || "0");

  let maclar = tumMaclar.filter(m => {
    if (arama && !(m.ev_sahibi || "").toLowerCase().includes(arama) && !(m.deplasman || "").toLowerCase().includes(arama) && !(m.lig || "").toLowerCase().includes(arama)) return false;
    if (tarih && m.tarih !== tarih) return false;
    if (lig && (m.lig || "") !== lig) return false;
    if (durum && m.durum !== durum) return false;
    return true;
  });

  // Her maç için kombinasyon yüzdesini hesapla
  let analizliMaclar = maclar.map(m => {
    let kesifSonuc = null;
    if (combo) {
      kesifSonuc = kesifYuzdeHesapla(m, combo);
    }
    return { ...m, kesifSonuc };
  });

  // Min yüzde filtresi
  if (combo && minYuzde > 0) {
    analizliMaclar = analizliMaclar.filter(m => m.kesifSonuc && m.kesifSonuc.yuzde >= minYuzde);
  }

  // Yüzdeye göre sırala (yüksekten düşüğe)
  if (combo) {
    analizliMaclar.sort((a, b) => {
      const ya = a.kesifSonuc ? a.kesifSonuc.yuzde : -1;
      const yb = b.kesifSonuc ? b.kesifSonuc.yuzde : -1;
      return yb - ya;
    });
  }

  kesifStatsGoster(analizliMaclar, combo);
  kesifRender(analizliMaclar, combo);
}

function kesifStatsGoster(maclar, combo) {
  const bi = maclar.filter(x => x.durum === "bitti").length;
  const comboAd = document.querySelector('#kesifKombinasyon option[value="' + combo + '"]')?.textContent || combo;
  
  let html = '<div class="si"><div class="sv">' + maclar.length + '</div><div class="sl">Maç</div></div>';
  html += '<div class="si"><div class="sv">' + bi + '</div><div class="sl">Biten</div></div>';
  if (combo) {
    const yuksek = maclar.filter(m => m.kesifSonuc && m.kesifSonuc.yuzde >= 70).length;
    html += '<div class="si"><div class="sv" style="color:#00c853">' + yuksek + '</div><div class="sl">%70+ (' + comboAd + ')</div></div>';
  }
  document.getElementById("kesifStatsBar").innerHTML = html;
}

function kesifOH(v, m, key) {
  if (!v) return '<div class="oran-hucre"><div class="ov" style="color:#333">-</div></div>';
  const cls = dogruOran(m, key) ? " dogru" : "";
  return '<div class="oran-hucre' + cls + '"><div class="ov">' + v + '</div><div class="op">%' + pct(v) + '</div></div>';
}

function kesifRender(maclar, combo) {
  const container = document.getElementById("kesifContainer");
  if (!maclar.length) {
    container.innerHTML = '<div class="bos">Sonuç yok</div>';
    return;
  }

  const grp = {};
  maclar.forEach(m => {
    if (!grp[m.tarih]) grp[m.tarih] = [];
    grp[m.tarih].push(m);
  });

  let h = "";
  Object.keys(grp).sort().forEach(tarih => {
    const gm = grp[tarih];
    const d = new Date(tarih);
    h += '<div class="gun-baslik">' + d.toLocaleDateString("tr-TR", { day: "numeric", month: "long", year: "numeric", weekday: "long" }) + '<span class="gc">' + gm.length + '</span></div>';
    
    h += '<div class="tablo-header">';
    h += '<div>Saat</div><div style="text-align:left">Maç</div>';
    h += '<div>MS1</div><div>MSX</div><div>MS2</div><div>ALT</div><div>ÜST</div><div>KGV</div><div>KGY</div>';
    if (combo) h += '<div style="background:#0d2818;color:#00c853;font-weight:700">' + combo.replace("_", " + ") + '</div>';
    h += '</div>';

    gm.forEach(m => {
      const o = m.oranlar || {};
      const skorTxt = m.durum === "bitti" ? ' <span style="color:#f0b90b;font-weight:700"> MS: ' + (m.skor_ev || 0) + '-' + (m.skor_dep || 0) + '</span>' : '';
      
      h += '<div class="mac-satir">';
      h += '<div class="ms-saat"><span class="s">' + (m.saat || "-") + '</span><span class="t">#' + (m.mac_kodu || "") + '</span></div>';
      h += '<div class="ms-takim"><span class="ev">' + (m.ev_sahibi || "?") + skorTxt + '</span><span class="dep">' + (m.deplasman || "?") + '</span></div>';
      
      h += kesifOH(o["Maç Sonucu_1"], m, "Maç Sonucu_1");
      h += kesifOH(o["Maç Sonucu_0"], m, "Maç Sonucu_0");
      h += kesifOH(o["Maç Sonucu_2"], m, "Maç Sonucu_2");
      h += kesifOH(o["Alt/Üst 2.5_Alt"], m, "Alt/Üst 2.5_Alt");
      h += kesifOH(o["Alt/Üst 2.5_Üst"], m, "Alt/Üst 2.5_Üst");
      h += kesifOH(o["Karşılıklı Gol_Var"], m, "Karşılıklı Gol_Var");
      h += kesifOH(o["Karşılıklı Gol_Yok"], m, "Karşılıklı Gol_Yok");

      if (combo) {
        const ks = m.kesifSonuc;
        if (ks) {
          let renk = "#e74c3c";
          let bg = "#2d1215";
          if (ks.yuzde >= 70) { renk = "#00c853"; bg = "#063b22"; }
          else if (ks.yuzde >= 50) { renk = "#f0b90b"; bg = "#2d2a0a"; }
          
          h += '<div class="oran-hucre" style="background:' + bg + ';border-left:2px solid ' + renk + '">';
          h += '<div class="ov" style="color:' + renk + ';font-size:13px;font-weight:700">%' + ks.yuzde + '</div>';
          h += '<div class="op" style="color:' + renk + ';opacity:0.7">' + ks.tutan + '/' + ks.toplam + '</div>';
          h += '</div>';
        } else {
          h += '<div class="oran-hucre"><div class="ov" style="color:#444">-</div><div class="op">veri yok</div></div>';
        }
      }

      h += '</div>';
    });
  });

  container.innerHTML = h;
}

function kesifReset() {
  document.getElementById("kesifArama").value = "";
  document.getElementById("kesifTarih").value = "";
  document.getElementById("kesifLig").value = "";
  document.getElementById("kesifDurum").value = "";
  document.getElementById("kesifKombinasyon").value = "";
  document.getElementById("kesifMinYuzde").value = "0";
  kombineKesifFiltrele();
}

// Tarih ve lig dropdown'larını kesif için de doldur
const orjYukle = yukle;
yukle = function() {
  orjYukle();
  setTimeout(() => {
    const tarihler = [...new Set(tumMaclar.map(m => m.tarih))].sort();
    const ks = document.getElementById("kesifTarih");
    if (ks) {
      tarihler.forEach(t => {
        const o = document.createElement("option");
        o.value = t;
        o.textContent = new Date(t).toLocaleDateString("tr-TR", { day: "numeric", month: "long", weekday: "short" });
        ks.appendChild(o);
      });
    }
    const ligler = [...new Set(tumMaclar.map(m => m.lig || "").filter(x => x.trim()))].sort();
    const kl = document.getElementById("kesifLig");
    if (kl) {
      ligler.forEach(l => {
        const o = document.createElement("option");
        o.value = l;
        o.textContent = l;
        kl.appendChild(o);
      });
    }
  }, 1500);
};
```

### Ne Yaptık?

1. **Bülten Benzeri Tablo:** Sol tarafta Saat, Maç, MS1, MSX, MS2, ALT, ÜST, KGV, KGY oranları (aynı Bülten'deki gibi).
2. **Sağda Kombinasyon Sütunu:** Dropdown'dan kombinasyon seçtiğinde (örn: MS1 + KG VAR), her maçın yanında o kombinasyonun geçmişte tutma yüzdesi görünüyor.
3. **Her Maç Kendini Analiz Ediyor:** Maçın MS1 oranı 1.85 ise, geçmişte 1.70-2.00 aralığında MS1 oranı açılan maçlarda "MS1 + KG VAR" ne sıklıkla tutmuş, bunu hesaplayıp gösteriyor.
4. **Renk Kodlaması:**
   - 🟢 **Yeşil** = %70+ (Güçlü)
   - 🟡 **Sarı** = %50-69 (Orta)
   - 🔴 **Kırmızı** = %50 altı (Zayıf)
5. **Sıralama:** En yüksek yüzdeli maçlar en üstte.
