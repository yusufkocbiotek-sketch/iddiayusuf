    # ---------------- TEMIZLIK ----------------
    def cleanup_old_profiles(self, max_age_hours=6):
        """Belirli bir yasindan eski chrome profil klasorlerini siler."""
        try:
            now = time.time()
            for item in os.listdir(CHROME_PROFILES_DIR):
                item_path = os.path.join(CHROME_PROFILES_DIR, item)
                if os.path.isdir(item_path):
                    mtime = os.path.getmtime(item_path)
                    age_hours = (now - mtime) / 3600.0
                    if age_hours > max_age_hours:
                        self.log(f"Eski Chrome profili siliniyor: {item_path}")
                        shutil.rmtree(item_path, ignore_errors=True)
        except Exception as e:
            self.log(f"UYARI: Eski Chrome profilleri temizlenirken hata: {e}")

    # ---------------- ULTRASURF ----------------
    def kill_ultrasurf(self):
        """Sistemde calisan tum Ultrasurf sureclerini sonlandirir."""
        for proc in psutil.process_iter(['pid', 'name']):
            try:
                name = (proc.info.get('name') or '').lower()
                if 'ultrasurf' in name:
                    self.log(f"Ultrasurf sonlandiriliyor (PID: {proc.info['pid']})")
                    proc.kill()
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                pass
            except Exception as e:
                self.log(f"UYARI: Ultrasurf sonlandirilirken hata: {e}")

    def is_proxy_port_open(self, host="127.0.0.1", port=9666, timeout=1):
        """Proxy portunun acik olup olmadigini kontrol eder."""
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                sock.settimeout(timeout)
                res = sock.connect_ex((host, port))
                return res == 0
        except Exception:
            return False

    def start_ultrasurf(self):
        """Ultrasurf'u baslatir ve proxy'nin hazir olmasini bekler."""
        self.kill_ultrasurf()

            self.log(f"HATA: Ultrasurf bulunamadi: {self.ultrasurf_path}")

        try:

            self.log(f"HATA: Ultrasurf baslatilamadi: {e}")
            return False

        self.log("Proxy portu bekleniyor (35 saniye)")

                self.log("ULTRASURF PROXY HAZIR")
                time.sleep(3)
                return True
            time.sleep(1)

        self.log("UYARI: Proxy portu acilmadi")
        return False

    def page_has_blocker(self, driver):

    def extract_match_name_from_row(self, row_element):
        """Bir mac satirindan takim isimlerini cikarir."""
        team_elems = []
        for sel in TEAM_SELECTORS:
            try:
                team_elems = row_element.find_elements(By.CSS_SELECTOR, sel)
                if len(team_elems) >= 2:
                    break
            except Exception:
                continue
        if len(team_elems) >= 2:
            team1 = team_elems[0].text.strip()
            team2 = team_elems[1].text.strip()
            if team1 and team2:
                return f"{team1} - {team2}"
        try:
            text = row_element.text.strip()
            m = re.search(r"([^\n\-]+?)\s*(?:-|vs)\s*([^\n\-]+)", text, re.IGNORECASE)
            if m:
                t1 = m.group(1).strip()
                t2 = m.group(2).strip()
                if t1 and t2:
                    return f"{t1} - {t2}"
            parts = [p.strip() for p in text.splitlines() if p.strip()]
            if len(parts) >= 2:
                return f"{parts[0]} - {parts[1]}"
        except Exception:
            pass
        return ""

    def is_valid_odds(self, text):
        try:
            num = float(text.strip().replace(",", "."))
            return 1.01 < num < 100.0 # Oran üst sınırı 20'den 100'e çıkarıldı
        except Exception:
            return False

    def extract_odds_count_from_page(self, driver, match_name_for_debug, match_url_for_debug, max_wait=35, retries=3):
        """
        Detay sayfasindan gecerli oran sayisini ceker.
        - Explicit wait ile oran konteynerini bekler
        - retries kadar tekrar deneme (bekle + refresh + farkli selector seti)
        """
        last_error = None
        for attempt in range(retries + 1):
            try:
                self.log(f"🔎 Oran çekme denemesi {attempt + 1}/{retries + 1} ({match_name_for_debug})")
                
                # Sayfada Human-like hareketler ve scroll
                self.human_like(driver)
                self.controlled_scroll(driver)
                time.sleep(random.uniform(1.0, 2.0)) # Küçük bir bekleme

                # 1) Oran konteynerini bekle (herhangi bir gecerli oran elementi gorunene kadar)
                WebDriverWait(driver, max_wait).until(
                    lambda d: any(
                        self.is_valid_odds(el.text.strip())
                        for sel in ODDS_SELECTORS
                        for el in d.find_elements(By.CSS_SELECTOR, sel)
                    )
                )
                self.log(f"✅ Oran elementleri sayfada yüklendiği görüldü ({match_name_for_debug}).")

                # 2) Kisa JS dolum beklemesi (daha uzun tutuldu)
                time.sleep(random.uniform(2.0, 4.0)) 
                
                # 3) Oranlari topla
                all_odds_vals = []
                for sel in ODDS_SELECTORS:
                    try:
                        elems = driver.find_elements(By.CSS_SELECTOR, sel)
                        for el in elems:
                            txt = el.text.strip()
                            if self.is_valid_odds(txt):
                                all_odds_vals.append(float(txt.replace(",", ".")))
                    except Exception:
                        continue

                # 4) Fallback: sayfa kaynak kodundan regex
                if len(all_odds_vals) < 5: # Çok az oran varsa regex'i dene
                    self.log(f"Yetersiz oran ({len(all_odds_vals)}), kaynak kodu taranıyor...")
                    try:
                        page_source = driver.page_source
                        # Regex'i daha esnek hale getir (örneğin 1.5, 1,5, 1.50)
                        found = re.findall(r"([1-9]\d*[,.]\d{1,2})", page_source)
                        for f in found:
                            fv = float(f.replace(",", "."))
                            if 1.01 < fv < 100.0:
                                all_odds_vals.append(fv)
                    except Exception:
                        pass

                # 5) Benzersiz oran sayisi
                unique_odds = set()
                for o in all_odds_vals:
                    unique_odds.add(round(o, 2))
                odds_count = len(unique_odds)

                if odds_count > 0: # <-- BURASI DEĞİŞTİRİLDİ (daha önce > 5 idi)
                    self.log(f"DETAY ORANLAR: {odds_count} adet bulundu ({match_name_for_debug})")
                    return odds_count
                else:
                    self.log(f"YETERSİZ/HİÇ ORAN BULUNAMADI ({odds_count}) - Deneme {attempt + 1}/{retries + 1} ({match_name_for_debug})")

            except TimeoutException as e:
                last_error = e
                self.log(f"Oran konteyneri zaman aşımı - Deneme {attempt + 1}/{retries + 1} ({match_name_for_debug})")
            except Exception as e:
                last_error = e
                self.log(f"HATA (oran çekme) - Deneme {attempt + 1}/{retries + 1}: {str(e)[:80]} ({match_name_for_debug})")

            # Tekrar denemeden once aksiyonlar
            if attempt < retries:
                time.sleep(random.uniform(3.0, 6.0)) # Daha uzun bekle
                try:
                    driver.refresh()
                    self.log(f"Sayfa yenileniyor... (Deneme {attempt+1}/{retries+1})")
                    time.sleep(random.uniform(3.0, 5.0)) # Yenileme sonrası bekle
                    self.human_like(driver) # Yeniden yüklenen sayfada da human-like davran
                except Exception as refresh_err:
                    self.log(f"UYARI: Sayfa yenilemede hata: {refresh_err}")
        
        # Tüm denemeler başarısız olursa HTML'i kaydet
        if driver:
            sanitized_name = re.sub(r'[^a-zA-Z0-9_\-]', '_', match_name_for_debug[:40]).strip('_')
            self.save_debug_html(driver, f"iddaa_fail_odds_{sanitized_name}")

        self.log(f"BAŞARISIZ: {match_name_for_debug} için detay oranlar boş (son hata: {last_error})")
        return 0
