    # ---------------- ULTRASURF ----------------
    def kill_ultrasurf(self):
        """Sistemde calisan tum Ultrasurf sureclerini sonlandirir."""
        for proc in psutil.process_iter(['pid', 'name']):
            try:
                name = (proc.info.get('name') or '').lower()
                if 'ultrasurf' in name:
                    self.log(f"Ultrasurf sonlandiriliyor (PID: {proc.info['pid']})")
                    proc.kill()
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                pass

    def is_proxy_port_open(self, host="127.0.0.1", port=9666, timeout=1):
        """Proxy portunun acik olup olmadigini kontrol eder."""
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                sock.settimeout(timeout)
                res = sock.connect_ex((host, port))
                return res == 0
        except Exception:
            return False

    def start_ultrasurf(self):
        """Ultrasurf'u baslatir ve proxy'nin hazir olmasini bekler."""
        self.kill_ultrasurf()
        self.log("ULTRASURF PROXY BASLATILIYOR")

        if not os.path.exists(self.ultrasurf_path):
            self.log(f"HATA: Ultrasurf bulunamadi: {self.ultrasurf_path}")
            return False

        try:
            # shell=True KULLANMA (guvenli calistirma)
            subprocess.Popen([self.ultrasurf_path])

            return False

        self.log("Proxy portu bekleniyor (35 saniye)")

                self.log("ULTRASURF PROXY HAZIR")
                time.sleep(3)  # stabilizasyon
                return True
            time.sleep(1)

        self.log("UYARI: Proxy portu acilmadi")
        return False

    def stealth_chrome_options(self):
        """Bot tespitini zorlastiran Chrome ayarlari."""
        options = Options()
        # Temel
        options.add_argument("--no-sandbox")
        options.add_argument("--disable-dev-shm-usage")
        options.add_argument("--disable-blink-features=AutomationControlled")
        options.add_experimental_option("excludeSwitches", ["enable-automation"])
        options.add_experimental_option("useAutomationExtension", False)
        # Ek
        options.add_argument("--disable-extensions")
        options.add_argument("--mute-audio")
        options.add_argument("--start-maximized")
        options.add_argument("--window-size=1366,768")
        # Proxy (Ultrasurf)
        options.add_argument(f"--proxy-server=http://127.0.0.1:{self.ultrasurf_port}")
        # User-Agent
        options.add_argument(
            "--user-agent=Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
            "AppleWebKit/537.36 (KHTML, like Gecko) "
            "Chrome/121.0.0.0 Safari/537.36"
        )
        return options

    def init_driver(self):
        options = self.stealth_chrome_options()
        self.driver = webdriver.Chrome(options=options)
        # WebDriver tespitini zorlastir
        self.driver.execute_cdp_cmd(

                    Object.defineProperty(navigator, 'webdriver', { get: () => undefined });

                    Object.defineProperty(navigator, 'languages', { get: () => ['tr-TR','tr','en-US','en'] });
                """

    def quit_driver(self):

    def verify_proxy_ip(self, driver):
        """Proxy'nin calisip calismadigini kontrol icin IP'yi loglar."""
        try:
            driver.set_page_load_timeout(20)
            driver.get("https://httpbin.org/ip")
            time.sleep(2)

            self.log(f"CIKAN IP (proxy kontrol): {ip_text}")
            return True
        except Exception as e:
            self.log(f"IP kontrolu basarisiz: {e}")
            return False

    def page_has_blocker(self, driver):
        """Sayfada captcha/engel metni var mi kontrol eder."""
        try:
            page_text = driver.page_source.lower()
            blockers = [
                "captcha",
                "verify you are human",
                "access denied",
                "rate limit",
                "güvenlik doğrulaması",
                "please enable javascript",
                "cloudflare",
                "checking your browser",
                "just a moment",
            ]
            for b in blockers:
                if b in page_text:
                    return True
            return False
        except Exception:
            return False

    def extract_match_name_from_row(self, row_element):
        """Bir mac satirindan takim isimlerini cikarir."""
        # Takim elementlerini ara
        team_elems = []
        for sel in TEAM_SELECTORS:
            try:
                team_elems = row_element.find_elements(By.CSS_SELECTOR, sel)
                if len(team_elems) >= 2:
                    break
            except Exception:
                continue
        # Takimlari al
        if len(team_elems) >= 2:
            team1 = team_elems[0].text.strip()
            team2 = team_elems[1].text.strip()
            if team1 and team2:
                return f"{team1} - {team2}"
        # Fallback: satir metninden regex ile
        try:
            text = row_element.text.strip()
            m = re.search(r"([^\n\-]+?)\s*(?:-|vs)\s*([^\n\-]+)", text, re.IGNORECASE)
            if m:
                t1 = m.group(1).strip()
                t2 = m.group(2).strip()
                if t1 and t2:
                    return f"{t1} - {t2}"
            parts = [p.strip() for p in text.splitlines() if p.strip()]
            if len(parts) >= 2:
                return f"{parts[0]} - {parts[1]}"
        except Exception:
            pass
        return ""

    def is_valid_odds(self, text):
        try:
            num = float(text.strip().replace(",", "."))
            return 1.01 < num < 20.0
        except Exception:
            return False

    def extract_odds_from_page(self, driver):
        """Sayfadan oranlari ceker (detay sayfasi icin)."""
        all_odds = []
        # Selector tabanli
        for sel in ODDS_SELECTORS:
            try:
                elems = driver.find_elements(By.CSS_SELECTOR, sel)
                for el in elems:
                    txt = el.text.strip()
                    if self.is_valid_odds(txt):
                        all_odds.append(float(txt.replace(",", ".")))
            except Exception:
                continue
        # Fallback: regex
        if len(all_odds) < 10:
            try:
                page_source = driver.page_source
                found = re.findall(r"([1-9]\d?\.\d{1,2})", page_source)
                for f in found:
                    fv = float(f)
                    if 1.01 < fv < 20.0:
                        all_odds.append(fv)
            except Exception:
                pass
        # Benzersiz
        unique = set()
        for o in all_odds:
            unique.add(round(o, 2))
        return len(unique)
