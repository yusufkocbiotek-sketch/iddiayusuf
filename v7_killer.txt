    # ---------------- TEMIZLIK ----------------
    def cleanup_old_profiles(self, max_age_hours=6):
        """Belirli bir yasindan eski chrome profil klasorlerini siler."""
        try:
            now = time.time()
            for item in os.listdir(CHROME_PROFILES_DIR):
                item_path = os.path.join(CHROME_PROFILES_DIR, item)
                if os.path.isdir(item_path):
                    mtime = os.path.getmtime(item_path)
                    age_hours = (now - mtime) / 3600.0
                    if age_hours > max_age_hours:
                        shutil.rmtree(item_path, ignore_errors=True)
        except Exception:
            pass

    # ---------------- ULTRASURF ----------------
    def kill_ultrasurf(self):
        """Sistemde calisan tum Ultrasurf sureclerini sonlandirir."""
        for proc in psutil.process_iter(['pid', 'name']):
            try:
                name = (proc.info.get('name') or '').lower()
                if 'ultrasurf' in name:
                    self.log(f"Ultrasurf sonlandiriliyor (PID: {proc.info['pid']})")
                    proc.kill()
            except (psutil.NoSuchProcess, psutil.AccessDenied):
                pass

    def is_proxy_port_open(self, host="127.0.0.1", port=9666, timeout=1):
        """Proxy portunun acik olup olmadigini kontrol eder."""
        try:
            with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
                sock.settimeout(timeout)
                res = sock.connect_ex((host, port))
                return res == 0
        except Exception:
            return False

    def page_has_blocker(self, driver):
        """Sayfada captcha/engel metni var mi kontrol eder."""
        try:
            page_text = driver.page_source.lower()
            blockers = [
                "captcha",
                "verify you are human",
                "access denied",
                "rate limit",
                "güvenlik doğrulaması",
                "please enable javascript",
                "cloudflare",
                "checking your browser",
                "just a moment",
                "ddos protection",
                "unusual traffic",
            ]
            for b in blockers:
                if b in page_text:
                    return True
            return False
        except Exception:
            return False

    def extract_match_name_from_row(self, row_element):
        """Bir mac satirindan takim isimlerini cikarir."""
        # Takim elementlerini ara
        team_elems = []
        for sel in TEAM_SELECTORS:
            try:
                team_elems = row_element.find_elements(By.CSS_SELECTOR, sel)
                if len(team_elems) >= 2:
                    break
            except Exception:
                continue
        # Takimlari al
        if len(team_elems) >= 2:
            team1 = team_elems[0].text.strip()
            team2 = team_elems[1].text.strip()
            if team1 and team2:
                return f"{team1} - {team2}"
        # Fallback: satir metninden regex ile
        try:
            text = row_element.text.strip()
            m = re.search(r"([^\n\-]+?)\s*(?:-|vs)\s*([^\n\-]+)", text, re.IGNORECASE)
            if m:
                t1 = m.group(1).strip()
                t2 = m.group(2).strip()
                if t1 and t2:
                    return f"{t1} - {t2}"
            parts = [p.strip() for p in text.splitlines() if p.strip()]
            if len(parts) >= 2:
                return f"{parts[0]} - {parts[1]}"
        except Exception:
            pass
        return ""

    def is_valid_odds(self, text):
        try:
            num = float(text.strip().replace(",", "."))
            return 1.01 < num < 20.0
        except Exception:
            return False

    def extract_odds_count_from_page(self, driver, max_wait=25, retries=2):
        """
        Detay sayfasindan gecerli oran sayisini ceker.
        - Explicit wait ile oran konteynerini bekler
        - retries kadar tekrar deneme (bekle + refresh + farkli selector seti)
        """
        last_error = None
        for attempt in range(retries + 1):
            try:
                # 1) Oran konteynerini bekle (herhangi bir gecerli oran elementi gorunene kadar)
                WebDriverWait(driver, max_wait).until(
                    lambda d: any(
                        self.is_valid_odds(el.text.strip())
                        for sel in ODDS_SELECTORS
                        for el in d.find_elements(By.CSS_SELECTOR, sel)
                    )
                )
                # 2) Kisa JS dolum beklemesi
                time.sleep(random.uniform(1.0, 2.0))

                # 3) Oranlari topla
                all_odds_vals = []
                for sel in ODDS_SELECTORS:
                    try:
                        elems = driver.find_elements(By.CSS_SELECTOR, sel)
                        for el in elems:
                            txt = el.text.strip()
                            if self.is_valid_odds(txt):
                                all_odds_vals.append(float(txt.replace(",", ".")))
                    except Exception:
                        continue

                # 4) Fallback: sayfa kaynak kodundan regex
                if len(all_odds_vals) < 10:
                    try:
                        page_source = driver.page_source
                        found = re.findall(r"([1-9]\d?\.\d{1,2})", page_source)
                        for f in found:
                            fv = float(f)
                            if 1.01 < fv < 20.0:
                                all_odds_vals.append(fv)
                    except Exception:
                        pass

                # 5) Benzersiz oran sayisi
                unique_odds = set()
                for o in all_odds_vals:
                    unique_odds.add(round(o, 2))
                odds_count = len(unique_odds)

                if odds_count > 5:
                    self.log(f"DETAY ORANLAR: {odds_count} adet bulundu")
                    return odds_count
                else:
                    self.log(f"YETERSIZ ORAN ({odds_count}) - deneme {attempt + 1}/{retries + 1}")

            except TimeoutException as e:
                last_error = e
                self.log(f"Oran konteyneri zaman asimi - deneme {attempt + 1}/{retries + 1}")
            except Exception as e:
                last_error = e
                self.log(f"HATA (oran cekme) - deneme {attempt + 1}/{retries + 1}: {str(e)[:80]}")

            # Tekrar denemeden once aksiyonlar
            if attempt < retries:
                time.sleep(random.uniform(2.5, 4.5))
                # Sayfayi yenilemeyi dene
                try:
                    driver.refresh()
                    time.sleep(random.uniform(2.0, 3.5))
                except Exception:
                    pass

        self.log(f"BASARISIZ: Detay oranlar bos (son hata: {last_error})")
        return 0
